% =======================================
\begin{frame}{Motivation}
\hypertarget{motivation_relevance}{}
\textbf{To achieve the energy transition, (residential) PV plays a vital role...}
\vspace{0.1cm}
\begin{itemize}
    \item Surpass coal in 2027 with 2'350 GW (22\%) of global power production capacity (\cite{iea_resreport2022})
    \item Has a potential of 67 TWh/y on existing Swiss infrastructure (\cite{vese_pvpower2021})
    \item Targeted to have 37.5 GW capacity in Switzerland in 2050 (\cite{bfe_ep2050_2020}, 8.17 GW in 2024 \cite{swisssolar2024}, \hyperlink{appendix_res_targets}{\beamerreturnbutton{App.Res Targets}})
\end{itemize}
\vspace{0.4cm }
\textbf{... but faces serious problems.}
\vspace{0.1cm}
\begin{itemize}
    \item Recent political opposition. 
    \item Dislike for large scale installations (\cite{sutterlin2017}, \cite{cousse2020})
    \item Grid overloading has already been observed and requires costly counter-measures (\cite{renewables2023}, \cite{tomar2024})
\end{itemize}
\end{frame}

\section{Outlook} 

% =======================================
\begin{frame}{Conclusion}
% \textbf{Summary}
\begin{block}{Research Contribution}
    \begin{enumerate}
        \item \textbf{Problem quantification}: 
        \begin{itemize}
            % \item Create highly detailed spatial model to estimate PV-induced effects in larger LV grid (incl. roof specifics + individual "decision making").
            %(incl. roof specifics production profiles + individual "decision making").
            \item \textcolor{unibasgreydark}{\textit{Problems will arise in the next couple of years and scale up fast}}
        \end{itemize}
        \vspace{0.1cm}
        \item \textbf{Economic response}: 
        \begin{itemize}
            % \item \textcolor{unibasgreydark}{1st-best) Congestion pricing for externality (not feasible in practice)}
            % \item 2nd-best) Simple subsidy adjustments, focusing on grid capacity
            \item \textcolor{unibasgreydark}{\textit{Simple subsidy adjustments can delay grid congestion by a few years and overall size, possibly maybe even as good as a social planners allocation. }}
        \end{itemize}
    \end{enumerate}
\end{block}

    \item Add battery investment and individual battery usage to the model 
    $\rightarrow$ optimize household consumption
    \item  More sophisticated pricing schemes (actual congestion pricing)
    \item Use RTP instead of TOU prices for electricity self consumption.
\end{itemize}
